




name: Setup Nginx with custom config
hosts: all
become: yes
tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: Copy custom nginx.conf
      copy:
        src: nginx.conf
        dest: /etc/nginx/nginx.conf
     - name: Start nginx service
      services:
      name: nginx
      state: started
      enabled: yes
