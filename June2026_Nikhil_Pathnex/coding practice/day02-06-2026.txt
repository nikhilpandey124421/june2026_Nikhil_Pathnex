ansible - install and enable nginx

name: install and start Nginx on pathnex
hosts:all
become:yes
   



   tasks:
        -name: Instal nginx
         yum:
         name:nginx  
         state:present


        -name: Enable nginx
          service:
          name:nginx
          state:started
          enabled:yes
